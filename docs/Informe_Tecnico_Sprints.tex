\section{Resumen y alcance}
LogiLearn implementa el 100\% del sílabo MATG1001 con motores reales: tablas de verdad (clasificación tautología/contradicción/contingencia), inferencias con trazabilidad y contraejemplo $2^{N}$, cuantificadores $\forall$/$\exists$ sobre $D\subset\mathbb{Z}$ con De Morgan y Δ, y conjuntos con Venn interactivo. Este informe documenta \textbf{qué} se construyó, \textbf{cómo} (arquitectura Vue 3 SFC, \texttt{src/lib/} aislado de \texttt{src/pages/}, \texttt{src/content/site.json:1} con \texttt{modoLiteral} por módulo) y \textbf{cuándo} (sprints 1→4). Fuentes unificadas: scratch \texttt{informe\_scratch\_*.pdf/.docx} de los 4 equipos + \texttt{TAREAS/04\_Deploy/sinergia/informe\_sinergia.tex:1} y \texttt{TAREAS/04\_Deploy/hijos-de-linus/informe\_los\_hijos\_de\_linus.tex:1} (PR30/32) + \texttt{TAREAS/hijos-de-linus/Instrucciones.md:1} (PR33). export function clasificarProposicion(nodo: NodoExpresion): {clasificacion: Clasificacion; filas: Fila[]}
const vars = recolectarVariables(nodo); // P,Q,R
const filas = generarFilas(nodo, vars); // 2^n asignaciones
const vals = filas.map(f=>f.resultado);
return vals.every(v=>v) ? 'tautologia' : vals.every(v=>!v) ? 'contradiccion' : 'contingencia';

\begin{lstlisting}[caption={engine — evaluarCuantificador}]
export function evaluarCuantificador(tipo: 'forall'|'exists', dominio: string[], fn:(x:string)=>boolean): Resultado {
  const traz = dominio.map(x=>({elemento:x, resultado: fn(x)}));
  const ok = tipo==='forall' ? traz.every(t=>t.resultado) : traz.some(t=>t.resultado);
  return {resultado: ok, trazabilidad: traz, deMorgan: {original, negado}};
}
\end{lstlisting}

\subsection{Conjuntos — Linus}

\begin{lstlisting}[caption={sets — potencia}]
export function potencia<T>(c: Set<T>): Set<Set<T>> {
  const arr=[...c]; const n=arr.length; const out=new Set<Set<T>>();
  for(let m=0;m<1<<n;m++){ const s=new Set<T>(); for(let i=0;i<n;i++) if(m>>i &1) s.add(arr[i]); out.add(s); }
  return out; // 2^n subconjuntos
}
\end{lstlisting}
